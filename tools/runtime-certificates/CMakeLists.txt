cmake_minimum_required(VERSION 3.16)
project(AquaRuntimeCertificates C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
if(NOT DEFINED MBEDTLS_SOURCE)
  message(FATAL_ERROR "Set MBEDTLS_SOURCE to pinned Mbed TLS 3.6.6 checkout")
endif()
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
set(IDENTITY "${CMAKE_CURRENT_LIST_DIR}/../../aqua_hatch/firmware/libraries/AquaIdentity/src")
add_executable(test-runtime-certificates ../test_runtime_certificates.cpp "${IDENTITY}/AquaCertificates.cpp")
target_include_directories(test-runtime-certificates PRIVATE "${IDENTITY}")
target_link_libraries(test-runtime-certificates PRIVATE mbedx509 mbedcrypto)
if(DEFINED ARDUINOJSON_SOURCE)
  add_executable(test-first-boot-identity ../test_first_boot_identity.cpp "${IDENTITY}/AquaCertificates.cpp" "${IDENTITY}/AquaIdentity.cpp")
  target_include_directories(test-first-boot-identity PRIVATE stubs "${IDENTITY}" "${ARDUINOJSON_SOURCE}/src")
  target_compile_definitions(test-first-boot-identity PRIVATE ARDUINOJSON_ENABLE_ARDUINO_STRING=1 ARDUINOJSON_ENABLE_ARDUINO_STREAM=0 ARDUINOJSON_ENABLE_ARDUINO_PRINT=0)
  target_link_libraries(test-first-boot-identity PRIVATE mbedx509 mbedcrypto)
  if(MSVC)
    target_compile_options(test-first-boot-identity PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(test-first-boot-identity PRIVATE -UNDEBUG)
  endif()
endif()
# Assertions are part of this verification tool, including in Release builds.
if(MSVC)
  target_compile_options(test-runtime-certificates PRIVATE /UNDEBUG)
else()
  target_compile_options(test-runtime-certificates PRIVATE -UNDEBUG)
endif()
